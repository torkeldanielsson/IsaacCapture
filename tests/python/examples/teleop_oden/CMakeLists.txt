# teleop_oden tests. Each test_*.py registers as its own ctest entry under the
# ``teleop_oden`` label; the ROS-dependent ones skip without rclpy.

file(GLOB TEST_FILES
    RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/test_*.py"
)

foreach(test_file ${TEST_FILES})
    get_filename_component(test_name "${test_file}" NAME_WE)
    add_test(
        NAME "teleop_oden_${test_name}"
        COMMAND uv run --python ${ISAAC_TELEOP_PYTHON_VERSION} --extra dev
            pytest -v --tb=short "${test_file}"
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    )
    set_tests_properties("teleop_oden_${test_name}" PROPERTIES
        LABELS "teleop_oden"
    )
endforeach()
